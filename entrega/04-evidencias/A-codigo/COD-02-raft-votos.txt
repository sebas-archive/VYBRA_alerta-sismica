  139  void Raft::alVoteReq(const Msg& m, uint64_t ahoraMs) {
  140    bool concedo = m.termino == termino_ && (votadoPor_ == 0 || votadoPor_ == m.de) &&
  141                   logAlDia(m.last_idx, m.last_term);
  142    if (concedo) {
  143      bool nuevo = votadoPor_ != m.de;
  144      votadoPor_ = m.de;
  145      guardar();  // antes de responder
  146      reiniciarTemporizador(ahoraMs);
  147      if (nuevo) suceso("voto", servidor() + " votó por Servidor " + std::to_string(m.de));
  148    }
  149    Msg r(Tipo::VoteResp);
  150    r.concedido = concedo;
  151    enviar(m.de, r);
  152  }

  154  void Raft::alVoteResp(const Msg& m, uint64_t ahoraMs) {
  155    if (rol_ != Rol::Candidato || m.termino != termino_ || !m.concedido) return;
  156    votos_.insert(m.de);
  157    if ((int)votos_.size() >= mayoria()) ganar(ahoraMs);
  158  }
