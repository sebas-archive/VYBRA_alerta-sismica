   36  export function proyectar(puntos, ancho, alto, margen = 24) {
   37    const pts = puntos.filter((p) => p && Number.isFinite(p.lat) && Number.isFinite(p.lon));
   38    let c = { lat: 0, lon: 0 };
   39    let minLat = 0, maxLat = 0, minLon = 0, maxLon = 0;
   40    if (pts.length) {
   41      minLat = Math.min(...pts.map((p) => p.lat)); maxLat = Math.max(...pts.map((p) => p.lat));
   42      minLon = Math.min(...pts.map((p) => p.lon)); maxLon = Math.max(...pts.map((p) => p.lon));
   43      c = { lat: (minLat + maxLat) / 2, lon: (minLon + maxLon) / 2 };
   44    }
   45    const mLat = (R_TIERRA * Math.PI) / 180;
   46    const mLon = mLat * Math.cos(rad(c.lat));
   47    const extX = (maxLon - minLon) * mLon;
   48    const extY = (maxLat - minLat) * mLat;
   49    const libreX = Math.max(1, ancho - 2 * margen);
   50    const libreY = Math.max(1, alto - 2 * margen);
   51    const metrosPorPx = Math.max(extX / libreX, extY / libreY, ENCUADRE_MIN_M / ancho);
   52    return {
   53      x: (p) => ancho / 2 + ((p.lon - c.lon) * mLon) / metrosPorPx,
   54      y: (p) => alto / 2 - ((p.lat - c.lat) * mLat) / metrosPorPx,
   55      metrosPorPx,
   56    };
   57  }

   59  // Zona del sismo: centroide ponderado por nivel de los nodos de la alerta que están ubicados y no se movieron.
   60  // radio = distancia máxima del centro a esos nodos, mínimo 30 m. Sin nodos válidos: null.
   61  export function zonaSismo(alerta, eventos, ubicaciones, movidos) {
   62    const ids = new Set(alerta?.eventos ?? []);
   63    const peso = new Map(); // nodo -> nivel mayor
   64    for (const ev of eventos ?? []) {
   65      if (!ids.has(ev.id)) continue;
   66      const u = ubicaciones?.[ev.nodo];
   67      if (!u || !Number.isFinite(u.lat) || !Number.isFinite(u.lon) || movidos?.has(ev.nodo)) continue;
   68      peso.set(ev.nodo, Math.max(peso.get(ev.nodo) ?? 0, Number(ev.nivel) || 0));
   69    }
   70    if (!peso.size) return null;
   71    const nodos = [...peso.keys()].sort((a, b) => a - b);
   72    let suma = 0, lat = 0, lon = 0;
   73    for (const n of nodos) { const w = peso.get(n) || 1e-9; suma += w; lat += ubicaciones[n].lat * w; lon += ubicaciones[n].lon * w; }
   74    const centro = { lat: lat / suma, lon: lon / suma };
   75    const radioM = Math.max(RADIO_MIN_M, ...nodos.map((n) => distanciaM(centro, ubicaciones[n])));
   76    return { lat: centro.lat, lon: centro.lon, radioM, nodos };
   77  }
