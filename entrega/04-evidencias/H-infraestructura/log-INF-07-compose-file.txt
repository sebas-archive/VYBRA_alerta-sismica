# Sistema completo sin hardware. Cada ESP32 (nodo) tiene asignado un servidor con su propia base de datos:
#   nodoN  = ESP32 simulado (toma y transporta datos; no guarda nada, como el real)
#   servidorN + dbN = servidor de datos (Fedora / openSUSE / Ubuntu) con su PostgreSQL y su volumen
#   panel  = ADMIN y CLIENTE; lee el estado de los nodos y el histórico de los servidores
# Windows Server no: sus contenedores solo funcionan en un host Windows.
#   docker compose up --build
#   Panel:   http://localhost:8095/        ADMIN: /admin/        Vista pública: /publico
# ESP32 REALES (sin tocar código): copia ejemplo.env a .env y pon las direcciones de tus ESP32:
#   NODO1_URL=http://192.168.x.x  NODO2_URL=http://192.168.x.x  NODO3_URL=http://192.168.x.x
# (o http://sismo-nodo-N.local si tu red resuelve mDNS; dentro de un contenedor suele hacer falta la IP).
# El panel toma las mismas NODOn_URL (NODOS solo hace falta para sobrescribirlas). Arranca sin los simulados
# (incluye `controlador` para apagar/encender desde ADMIN; sin él la función queda desactivada):
#   docker compose up -d --build --no-deps panel servidor1 servidor2 servidor3 db1 db2 db3 controlador
# Comprueba antes que responden con el contrato esperado:   node web/scripts/verificar-esp32.js
# PRODUCCIÓN: docker compose -f docker-compose.yml -f docker-compose.prod.yml up -d --build   (ver docs/servidores-de-datos.md)
# Contraseña de las bases en DESARROLLO: DB_PASSWORD (por defecto "vybra"; nunca en producción). Las bases no publican puerto.
x-nodo: &nodo
  restart: unless-stopped
  healthcheck:
    test: ["CMD", "nodo-sim", "--salud"]
    interval: 5s
    timeout: 3s
    retries: 6
    start_period: 5s
x-servidor: &servidor
  restart: unless-stopped
  healthcheck:
    test: ["CMD", "node", "-e", "fetch('http://127.0.0.1:8100/salud').then(r=>process.exit(r.ok?0:1)).catch(()=>process.exit(1))"]
    interval: 5s
    timeout: 3s
    retries: 10
    start_period: 10s
# Retención/archivado (desactivado por defecto: no se archiva ni se elimina nada). Ver docs/servidores-de-datos.md.
x-retencion: &retencion
  RETENCION_ACTIVA: ${RETENCION_ACTIVA:-no}
  RETENCION_MODO: ${RETENCION_MODO:-archivar}
  RETENCION_CONFIRMAR_ELIMINAR: ${RETENCION_CONFIRMAR_ELIMINAR:-no}
  RETENCION_CRITICO_NIVEL: ${RETENCION_CRITICO_NIVEL:-6}
  RETENCION_MUESTRAS_DIAS: ${RETENCION_MUESTRAS_DIAS:-0}
  RETENCION_SUCESOS_DIAS: ${RETENCION_SUCESOS_DIAS:-0}
  RETENCION_LOG_DIAS: ${RETENCION_LOG_DIAS:-0}
  RETENCION_ACCIONES_DIAS: ${RETENCION_ACCIONES_DIAS:-0}
  RETENCION_EVENTOS_DIAS: ${RETENCION_EVENTOS_DIAS:-0}
  RETENCION_ALERTAS_DIAS: ${RETENCION_ALERTAS_DIAS:-0}
x-db: &db
  image: postgres:16-alpine
  restart: unless-stopped
  environment: &dbenv
    POSTGRES_USER: vybra
    POSTGRES_PASSWORD: ${DB_PASSWORD:-vybra}
    POSTGRES_DB: vybra
  healthcheck:
    test: ["CMD-SHELL", "pg_isready -U vybra -d vybra"]
    interval: 4s
    timeout: 3s
    retries: 15

services:
  # ---- ESP32 simulados: el MISMO núcleo Raft del firmware compilado para Linux (se hablan por UDP 4210) ----
  nodo1:
    <<: *nodo
    build:
      context: ./web
      dockerfile: docker/Dockerfile.nodo-sim
      additional_contexts:
        nucleo: ./firmware/detector/src
    container_name: ${PREFIJO:-sismo}-nodo-1
    environment: { N: "1", PUERTO_HTTP: "8080", PUERTO_RAFT: "4210" }
    ports: ["${PUERTO_NODO1:-9001}:8080"]
  nodo2:
    <<: *nodo
    build:
      context: ./web
      dockerfile: docker/Dockerfile.nodo-sim
      additional_contexts:
        nucleo: ./firmware/detector/src
    container_name: ${PREFIJO:-sismo}-nodo-2
    environment: { N: "2", PUERTO_HTTP: "8080", PUERTO_RAFT: "4210" }
    ports: ["${PUERTO_NODO2:-9002}:8080"]
  nodo3:
    <<: *nodo
    build:
      context: ./web
      dockerfile: docker/Dockerfile.nodo-sim
      additional_contexts:
        nucleo: ./firmware/detector/src
    container_name: ${PREFIJO:-sismo}-nodo-3
    environment: { N: "3", PUERTO_HTTP: "8080", PUERTO_RAFT: "4210" }
    ports: ["${PUERTO_NODO3:-9003}:8080"]

  # ---- Bases de datos (una por servidor) ----
  db1:
    <<: *db
    container_name: ${PREFIJO:-sismo}-db-1
    volumes: ["sismo-db1:/var/lib/postgresql/data"]
  db2:
    <<: *db
    container_name: ${PREFIJO:-sismo}-db-2
    volumes: ["sismo-db2:/var/lib/postgresql/data"]
  db3:
    <<: *db
    container_name: ${PREFIJO:-sismo}-db-3
    volumes: ["sismo-db3:/var/lib/postgresql/data"]

  # ---- Servidores de datos (cada uno en un sistema operativo distinto) ----
  servidor1:
    <<: *servidor
    build: { context: ./web, dockerfile: docker/Dockerfile.fedora }
    container_name: ${PREFIJO:-sismo}-servidor-1
    environment:
      <<: *retencion
      UV_THREADPOOL_SIZE: "32"
      N: "1"
      PUERTO: "8100"
      NODO_URL: ${NODO1_URL:-http://nodo1:8080}
      DATABASE_URL: postgres://vybra:${DB_PASSWORD:-vybra}@db1:5432/vybra
    ports: ["${PUERTO_SERVIDOR1:-8101}:8100"]
    depends_on: { db1: { condition: service_healthy } }
  servidor2:
    <<: *servidor
    build: { context: ./web, dockerfile: docker/Dockerfile.suse }
    container_name: ${PREFIJO:-sismo}-servidor-2
    environment:
      <<: *retencion
      UV_THREADPOOL_SIZE: "32"
      N: "2"
      PUERTO: "8100"
      NODO_URL: ${NODO2_URL:-http://nodo2:8080}
      DATABASE_URL: postgres://vybra:${DB_PASSWORD:-vybra}@db2:5432/vybra
    ports: ["${PUERTO_SERVIDOR2:-8102}:8100"]
    depends_on: { db2: { condition: service_healthy } }
  servidor3:
    <<: *servidor
    build: { context: ./web, dockerfile: docker/Dockerfile.ubuntu }
    container_name: ${PREFIJO:-sismo}-servidor-3
    environment:
      <<: *retencion
      UV_THREADPOOL_SIZE: "32"
      N: "3"
      PUERTO: "8100"
      NODO_URL: ${NODO3_URL:-http://nodo3:8080}
      DATABASE_URL: postgres://vybra:${DB_PASSWORD:-vybra}@db3:5432/vybra
    ports: ["${PUERTO_SERVIDOR3:-8103}:8100"]
    depends_on: { db3: { condition: service_healthy } }

  # ---- Controlador de infraestructura: único con acceso al socket de Docker; solo puede parar/arrancar nodoN y servidorN ----
  controlador:
    build: { context: ./web, dockerfile: docker/Dockerfile.controlador }
    container_name: ${PREFIJO:-sismo}-controlador
    restart: unless-stopped
    environment:
      PUERTO: "8200"
      CONTROL_TOKEN: ${CONTROL_TOKEN:-[TOKEN-DESARROLLO]}   # solo desarrollo; en producción CONTROL_TOKEN_FILE (secreto)
    volumes: ["/var/run/docker.sock:/var/run/docker.sock"]
    read_only: true
    cap_drop: [ALL]
    security_opt: ["no-new-privileges:true"]
    # Sin "ports": no se publica; solo el panel lo alcanza por la red interna de compose.
    healthcheck:
      test: ["CMD", "node", "-e", "fetch('http://127.0.0.1:8200/salud').then(r=>process.exit(r.ok?0:1)).catch(()=>process.exit(1))"]
      interval: 5s
      timeout: 3s
      retries: 6

  panel:
    build: ./web          # Alpine (node:22-alpine)
    container_name: ${PREFIJO:-sismo}-panel
    restart: unless-stopped
    command: ["node", "server.js"]
    environment:
      PUERTO: "8090"
      # Resolver el nombre de un contenedor APAGADO tarda unos 3 s (Docker lo reenvía al DNS del equipo): con el pool de hilos
      # por defecto (4) esas búsquedas retrasarían las peticiones a los nodos vivos.
      UV_THREADPOOL_SIZE: "32"
      NODOS: ${NODOS:-${NODO1_URL:-http://nodo1:8080},${NODO2_URL:-http://nodo2:8080},${NODO3_URL:-http://nodo3:8080}}
      SERVIDORES: http://servidor1:8100,http://servidor2:8100,http://servidor3:8100
      # Apagar/encender servidores desde ADMIN: el panel habla con el controlador (red interna), nunca con Docker.
      CONTROLADOR_URL: http://controlador:8200
      CONTROL_TOKEN: ${CONTROL_TOKEN:-[TOKEN-DESARROLLO]}
    ports: ["${PUERTO_PANEL:-8095}:8090"]
    depends_on:
      nodo1: { condition: service_healthy }
      nodo2: { condition: service_healthy }
      nodo3: { condition: service_healthy }
      servidor1: { condition: service_healthy }
      servidor2: { condition: service_healthy }
      servidor3: { condition: service_healthy }

  # HTTPS delante del panel: https://IP:8443/cliente/ y https://IP:8443/admin/ (ver docker/Caddyfile).
  # Va en el perfil "https" para no alterar el stack por defecto (11 servicios) ni chocar con el stack de
  # pruebas aislado; se arranca con: docker compose --profile https up -d
  https:
    profiles: ["https"]
    image: caddy:2-alpine
    container_name: ${PREFIJO:-sismo}-https
    restart: unless-stopped
    environment:
      HOST_PUBLICO: ${HOST_PUBLICO:-[IP-PC]}
    ports: ["${PUERTO_HTTPS:-8443}:8443"]
    volumes:
      - ./docker/Caddyfile:/etc/caddy/Caddyfile:ro
      - sismo-caddy:/data
    depends_on: [panel]

volumes:
  sismo-caddy:
  sismo-db1:
  sismo-db2:
  sismo-db3:
